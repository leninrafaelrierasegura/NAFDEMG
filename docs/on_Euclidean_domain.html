# Parameters
a <- 1
b <- 1
kappa <- 1
alpha <- 1.5
m = 1
beta <- alpha/2

nx <- 10
ny <- 10
T_final <- 1
time_step <- 0.1 
time_seq <- seq(0, T_final, by = time_step)
coeff <- 1
which_eig <- 4
FEM_matrices <- fem$fem_shifted_laplacian_sparse(
  a = a,
  b = b, 
  nx = nx, 
  ny = ny, 
  kappa = kappa
  )

L <- FEM_matrices$L
C <- FEM_matrices$C

true_eig <- compute_true_eigen_rectangle(
  a = a, 
  b = b, 
  nx = nx, 
  ny = ny, 
  kappa = kappa,
  m_max = 3, 
  n_max = 3
  )

x <- true_eig$x
y <- true_eig$y
eigvals <- true_eig$eigvals
eigfuncs <- true_eig$eigfuncs
U_0 <- coeff * eigfuncs[[which_eig]]
U_true <- lapply(exp(-eigvals[which_eig]^(alpha/2) * time_seq), function(x) x * U_0)
my_op_frac <- my.fractional.operators.frac(L, beta, C, scale.factor = kappa^2, m = m, time_step)


U_aux <- solve_fractional_evolution(my_op_frac, time_step, time_seq, val_at_0 = as.vector(U_0))

U_approx <- result_list <- lapply(seq_len(ncol(U_aux)), function(j) matrix(U_aux[, j], nx+1, ny+1))
plot_3d_slider(x, y, time_seq, U_true)
plot_3d_slider(x, y, time_seq, U_approx)
diff <- Map("-", U_true, U_approx)
plot_3d_slider(x, y, time_seq, diff)
