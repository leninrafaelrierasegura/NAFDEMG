To observe convergence in \(m\), we calibrate \(h\) and \(\tau\) to \(m\) as \(h \sim e^{-\pi\sqrt{(1-\alpha/2)m}}\) and then \(\tau \sim h^{\alpha}\), respectively. With this scaling, the convergence with respect to the rational order \(m\) is exponential. That is, \(\|u-U_{h,m}^\tau\|_{L_2((0,T);L_2(\Gamma))} \leq C_me^{-\alpha\pi\sqrt{(1-\alpha/2)m}}\). This can be verified by estimating the slope \(S_m\) in the regression \(\log_e E = S_m\sqrt{m} +\log_e C_m\), where we expect \(S_m\sim -\alpha\pi\sqrt{(1-\alpha/2)}\).

# Parameters
T_final <- 2
kappa <- 4
divider <- 1

L_max <- 10
rot.inv <- TRUE

coeff_U_0 <- rep(0, L_max+1)
coeff_U_0[9] <- 1
coeff_FF <- rep(0, L_max+1)
coeff_FF[2] <- 10

AAA = 1
OMEGA = pi

# Time step and mesh size
m_vector <- c(1, 2, 3, 4)
alpha_vector <- seq(1.6, 1.8, by = 0.2)
# Overkill parameters
overkill_time_step <- 0.01
overkill_h <- 0.05
globe_overkill <- globe_from_h(overkill_h)
# Finest time and space mesh
overkill_time_seq <- seq(0, T_final, length.out = ((T_final - 0) / overkill_time_step + 1))

mesh_overkill <- fm_rcdt_2d(globe = globe_overkill)
overkill_C <- fm_fem(mesh_overkill, order = 1)$c1
# Create a matrix to store the errors
errors_projected <- matrix(NA, nrow = length(m_vector), ncol = length(alpha_vector))
for (j in 1:length(alpha_vector)) {
  alpha <- alpha_vector[j] # from 0.5 to 2
  beta <- alpha / 2

  # Compute the eigenvalues and eigenfunctions on the finest mesh
  overkill_eigen_params <- compute_true_eigen_sphere(
    mesh = mesh_overkill, 
    kappa = kappa,
    alpha = alpha,
    L_max = L_max,
    rot.inv = rot.inv)
  EIGENVAL_ALPHA <- overkill_eigen_params$EIGENVAL_ALPHA # Eigenvalues (they are independent of the meshes)
  overkill_EIGENFUN <- overkill_eigen_params$EIGENFUN # Eigenfunctions on the finest mesh
  
  # Compute the true solution on the finest mesh
  overkill_U_true <- overkill_EIGENFUN %*% 
    outer(1:length(coeff_U_0), 
          1:length(overkill_time_seq), 
          function(i, j) (coeff_U_0[i] + coeff_FF[i] * G_sin(t = overkill_time_seq[j], A = AAA, lambda_j_alpha_half = EIGENVAL_ALPHA[i], omega = OMEGA)) * exp(-EIGENVAL_ALPHA[i] * overkill_time_seq[j]))
  
  for (i in 1:length(m_vector)) {
    m <- m_vector[i]
    h <- exp(- pi * sqrt((1 - alpha / 2) * m))/divider 
    time_step <- (h^alpha)
    globe <- globe_from_h(h)
    mesh <- fm_rcdt_2d(globe = globe)
    FEM <- fm_fem(mesh, order = 1)
    G <- FEM$g1
    C <- FEM$c1
    L <- kappa^2*C + G
    eigen_params <- compute_true_eigen_sphere(
    mesh = mesh, 
    kappa = kappa,
    alpha = alpha,
    L_max = L_max,
    rot.inv = rot.inv)
    EIGENFUN <- eigen_params$EIGENFUN
    U_0 <- EIGENFUN %*% coeff_U_0 # Compute U_0 on the current mesh
    Psi <- fm_basis(mesh, mesh_overkill$loc)
  
    time_seq <- seq(0, T_final, length.out = ((T_final - 0) / time_step + 1))
    my_op_frac <- my.fractional.operators.frac(L, beta, C, scale.factor = kappa^2, m = m, time_step)
    INT_BASIS_EIGEN <- t(overkill_EIGENFUN) %*% overkill_C %*% Psi
    # Compute matrix F with columns F^k
    FF_approx <- t(INT_BASIS_EIGEN) %*% 
      outer(1:length(coeff_FF), 
            1:length(time_seq), 
        function(i, j) coeff_FF[i] * g_sin(r = time_seq[j], A = AAA, omega = OMEGA))
    
    U_approx <- solve_fractional_evolution(my_op_frac, time_step, time_seq, val_at_0 = U_0, RHST = FF_approx)
    
    projected_U_approx <- Psi %*% U_approx
    projected_U_piecewise <- construct_piecewise_projection(projected_U_approx, time_seq, overkill_time_seq)
    XX <- overkill_U_true - projected_U_piecewise
    error_ij <-  sqrt(as.double(overkill_time_step * sum(XX * (overkill_C %*% XX)))) 
    errors_projected[i,j] <- error_ij
    slackr_msg(text = paste0("m =", m, ", alpha =", alpha, ", h =", h, ", time_step =", time_step, ", error =", error_ij), channel = "#research")
  }
}
save(errors_projected, file = here::here("data_files/errors_projected_m_sphere.RData"))
load(here::here("data_files/errors_projected_m_sphere.RData"))
observed_rates <- numeric(length(alpha_vector))
for (u in 1:length(alpha_vector)) {
  observed_rates[u] <- coef(lm(log(errors_projected[, u]) ~ sqrt(m_vector)))[2]
}

theoretical_rates <- - pi * alpha_vector * sqrt((1 - alpha_vector / 2))

p_m <- error.convergence.plotter(x_axis_vector = m_vector, 
                                 alpha_vector, 
                                 errors_projected, 
                                 theoretical_rates, 
                                 observed_rates,
                                 line_equation_fun = exp_line_equation,
                                 fig_title = expression("Convergence in " * italic(m)),
                                 x_axis_label = expression(italic(m)),
                                 apply_sqrt = TRUE)
p_m

Dashed lines indicate the theoretical rates, and solid lines represent the observed error curves. The legend below each plot shows the value of \(\alpha\) along with the corresponding theoretical (‘theo’), and observed (‘obs’) rates for each case.
