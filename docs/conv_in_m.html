To study convergence in \(m\), we keep the spatial mesh size \(h\) and time step \(\tau\) fixed. Theorem lastlasttheorem in paper/main.tex gives \[ E\lesssim\tau+h^{\alpha-\epsilon} +h^{\alpha-2}\exp\!\left(-2\pi\sqrt{(1-\alpha/2)m}\right), \qquad 0<\alpha<2. \] The rational contribution therefore has reference slope \(S_m=-2\pi\sqrt{1-\alpha/2}\) in a regression of \(\ln E\) against \(\sqrt{m}\). This is an upper-bound prediction for that contribution, rather than a guaranteed slope of the total error. Temporal, spatial, and reference-quadrature errors can produce a plateau as \(m\) increases; their sizes should be checked independently.

The physical equation is \(u_t+\nu L^{\alpha/2}u=f\), with diffusion \(=\nu>0\). Keep diffusion fixed along each refinement curve. The original meshes, package coefficients, interval-averaged forcing, and right-constant reference-time quadrature are retained. error_method = "compressed" evaluates the same reference-matrix norm using centered coefficient moments. It still advances every FEM/backward Euler step and needs no discrete eigenfunction identities. Use error_method = "reference" for the original accumulator. Execution settings (workers, error_n_blocks) do not change the mathematical problem.

# remotes::install_github("davidbolin/rspde", ref = "devel")
# remotes::install_github("davidbolin/metricgraph", ref = "devel")
library(rSPDE)
library(MetricGraph)
library(grateful)

library(ggplot2)
library(reshape2)
library(plotly)
# Results stay in the private run directory; no notification setup.
# Parameters
T_final <- 1e-3
diffusion <- 1 # nu > 0 in u_t + nu * L^(alpha/2) u = f
workers <- 4L # independent complete cases; set 1L for serial execution
error_method <- "compressed" # "reference" retains the original error accumulator
run_id <- NULL # use an existing run ID only to reload matching parameters
kappa <- 4
N_finite <- 4 # choose even
adjusted_N_finite <- N_finite + N_finite/2 + 1
# Coefficients for u_0 and f
coeff_U_0 <- 50*(1:adjusted_N_finite)^-1
coeff_U_0[-5] <- 0
coeff_FF <- rep(0, adjusted_N_finite)
coeff_FF[7] <- 10

AAA = 1
OMEGA = pi

# Vary only the rational order; keep the spatial and temporal grids fixed.
m_vector <- c(1, 2, 3, 4)
alpha_vector <- seq(0.5, 1.5, by = 0.4)
h_fixed <- 1e-4
time_step_target <- 5e-9
time_grid <- make_time_grid(T_final = T_final, target_step = time_step_target)
time_step <- time_grid$time_step
time_seq <- time_grid$time_seq
g_average <- average_sine_forcing(time_seq, A = AAA, omega = OMEGA)
# Reference quadrature and error accumulation.
overkill_time_step <- 1e-9
overkill_h <- 9e-5
error_verbose <- TRUE

study_parameters <- list(
  study = "m", T_final = T_final, diffusion = diffusion, kappa = kappa,
  N_finite = N_finite, initial = coeff_U_0, forcing = coeff_FF,
  A = AAA, omega = OMEGA, alpha = alpha_vector, axis = m_vector,
  fixed_h = h_fixed, time_step = time_step, reference_h = overkill_h,
  reference_time_step_target = overkill_time_step, error_method = error_method,
  coefficient_source = "rSPDE package",
  package_versions = vapply(c("Matrix", "MetricGraph", "rSPDE"),
                           function(p) as.character(utils::packageVersion(p)), character(1)))
run_dir <- reference_experiment_run("m", study_parameters, run_id)
result_file <- file.path(run_dir, "errors_projected_m.RData")
cat("Run ID:", basename(run_dir), "\nPrivate results:", run_dir, "\n")

overkill_graph <- gets.graph.tadpole(h = overkill_h)
overkill_graph$compute_fem()
overkill_C <- overkill_graph$mesh$C
error_n_blocks <- 1L
# Preview modeled memory before allocating the reference time vector or solving.
# estimated_gib excludes FEM/solver objects and other session memory by default.
error_memory <- estimate_error_memory(
  n_reference_nodes = nrow(overkill_C),
  n_exact_modes = if (error_method == "compressed")
    sum(coeff_U_0 != 0 | coeff_FF != 0) else NULL,
  n_reference_samples = ceiling(T_final / overkill_time_step),
  n_blocks = unique(c(error_n_blocks, 10, 100, 1000, 10000, 100000, 1000000)),
  n_solver_steps = time_grid$N
)
print(error_memory, digits = 3, row.names = FALSE)
# Set error_n_blocks above to the desired count before running the experiment.
reference_grid <- make_time_grid(
  T_final = T_final, target_step = overkill_time_step
)
overkill_time_step <- reference_grid$time_step
overkill_time_seq <- reference_grid$time_seq

# The spatial discretization is fixed for every rational order and alpha.
graph <- gets.graph.tadpole(h = h_fixed)
graph$compute_fem()
C <- graph$mesh$C
L <- kappa^2 * C + graph$mesh$G
Psi <- graph$fem_basis(overkill_graph$get_mesh_locations())
stopifnot(all(coeff_U_0[-5] == 0))
U_0 <- project_u0_tadpole(graph, amplitude = coeff_U_0[5])

# Reference spatial modes and loads are independent of alpha, tau, and m.
reference_eigen_params <- gets.eigen.params(
  N_finite = N_finite, kappa = kappa,
  alpha = alpha_vector[1], graph = overkill_graph
)
reference_eigenvalues <- reference_eigen_params$EIGENVAL
overkill_EIGENFUN <- reference_eigen_params$EIGENFUN
active_modes <- which(coeff_U_0 != 0 | coeff_FF != 0)
exact_basis <- overkill_EIGENFUN[, active_modes, drop = FALSE]
exact_initial <- coeff_U_0[active_modes]
exact_forcing <- coeff_FF[active_modes]
spatial_modes <- t(Psi) %*% overkill_C %*% overkill_EIGENFUN
spatial_load <- as.numeric(spatial_modes %*% coeff_FF)
error_projection <- if (error_method == "compressed")
  prepare_reference_error(Psi, overkill_C, exact_basis) else NULL
# Only the coordinator saves results; refuse overwriting an existing run.
stopifnot(!file.exists(result_file))
errors_projected <- matrix(NA_real_, length(m_vector), length(alpha_vector))
for (j in seq_along(alpha_vector)) {
  alpha <- alpha_vector[j]
  beta <- alpha / 2
  exact_rates <- diffusion * reference_eigenvalues[active_modes]^beta
  coefficients_at_time <- function(t) exact_modal_coefficients(
    t, exact_rates, exact_initial, exact_forcing, A = AAA, omega = OMEGA)
  exact_at_time <- function(t) exact_basis %*% coefficients_at_time(t)

  cases <- run_reference_cases(seq_along(m_vector), function(i) {
    m <- m_vector[i]
    my_op_frac <- my.fractional.operators.frac(
      L = L, beta = beta, C = C, scale.factor = kappa^2,
      m = m, time_step = time_step, diffusion = diffusion)
    solve_fractional_evolution_error(
      my_op_frac, time_step, time_seq, U_0, spatial_load, g_average,
      Psi, overkill_C, overkill_time_seq, exact_at_time,
      n_blocks = error_n_blocks, verbose = error_verbose,
      error_method = error_method, coefficients_at_time = coefficients_at_time,
      error_projection = error_projection)
  }, workers = workers)
  errors_projected[, j] <- vapply(cases, identity, numeric(1))
}
save(errors_projected, study_parameters, file = result_file)
saved <- new.env()
load(result_file, envir = saved)
stopifnot(identical(saved$study_parameters, study_parameters))
errors_projected <- saved$errors_projected
observed_rates <- numeric(length(alpha_vector))
for (u in 1:length(alpha_vector)) {
  observed_rates[u] <- coef(lm(log(errors_projected[, u]) ~ sqrt(m_vector)))[2]
}

theoretical_rates <- -2 * pi * sqrt(1 - alpha_vector / 2)

p_m_tadpole <- error.convergence.plotter(x_axis_vector = m_vector, 
                                 alpha_vector, 
                                 errors_projected, 
                                 theoretical_rates, 
                                 observed_rates,
                                 line_equation_fun = exp_line_equation,
                                 fig_title = expression("Convergence in " * italic(m)),
                                 x_axis_label = expression(italic(m)),
                                 apply_sqrt = TRUE)

p_m_nicer_tadpole <- errorConvergencePlotterLatex(x_axis_vector = m_vector, 
                                 alpha_vector, 
                                 errors_projected, 
                                 theoretical_rates, 
                                 observed_rates,
                                 line_equation_fun = exp_line_equation,
                                 fig_title = "$\\mbox{Convergence in }m$",
                                 x_axis_label = "$m\\quad(\\sqrt{\\phantom{x}}\\mbox{ scale})$",
                                 y_axis_label = "$\\|u-U_{h,m}^{\\tau}\\|_{L_2(\\Gamma\\times[0,T])}\\quad(\\log_{\\mathrm{e}}\\mbox{ scale})$",
                                 color_label = "$\\quad\\quad\\;\\;\\alpha\\;|\\;\\phantom{-}\\mathtt{theo}\\;|\\;\\;\\phantom{-}\\mathtt{obs}\\;\\;|\\%\\mathtt{re}$",
                                 apply_sqrt = TRUE)

ggsave(file.path(run_dir, "conv_rates_m_tadpole.png"), width = 4, height = 5, plot = p_m_tadpole, dpi = 300)
myggsave(p_m_nicer_tadpole, width = 4, height = 5, use_mathpazo = TRUE,
         output_dir = file.path(run_dir, "tikzpic"), copy_to_presentation = FALSE)
save(p_m_tadpole, file = file.path(run_dir, "p_m_tadpole.RData"))
save(p_m_nicer_tadpole, file = file.path(run_dir, "p_m_nicer_tadpole.RData"))
knitr::include_graphics(file.path(run_dir, "tikzpic", "p_m_nicer_tadpole_with_mathpazo.pdf"))
